//! CPU backend ("menggambar langsung menggunakan cpu"): a software rasteriser.
//! No dependencies: every primitive is blended into an RGBA buffer, then written as PNG.

use super::font;
use super::pixels::RgbaBuffer;
use super::Renderer;
use crate::component::{Rect, Rgba};

pub struct CpuRenderer {
    buffer: RgbaBuffer,
}

impl CpuRenderer {
    pub fn new() -> Self {
        Self { buffer: RgbaBuffer::new(1.0, 1.0) }
    }
}

impl Default for CpuRenderer {
    fn default() -> Self {
        Self::new()
    }
}

impl Renderer for CpuRenderer {
    type Error = String;

    fn begin(&mut self, width: f32, height: f32) -> Result<(), Self::Error> {
        self.buffer = RgbaBuffer::new(width, height);
        Ok(())
    }

    fn fill_rect(&mut self, rect: Rect, fill: Rgba, radius: f32, opacity: f32) -> Result<(), Self::Error> {
        self.buffer.rect(rect, fill, radius, opacity);
        Ok(())
    }

    fn stroke_rect(&mut self, rect: Rect, stroke: Rgba, width: f32, radius: f32) -> Result<(), Self::Error> {
        let w = width.max(1.0);
        self.buffer.rect(Rect { x: rect.x, y: rect.y, width: rect.width, height: w }, stroke, radius, 1.0);
        self.buffer.rect(Rect { x: rect.x, y: rect.y + rect.height - w, width: rect.width, height: w }, stroke, radius, 1.0);
        self.buffer.rect(Rect { x: rect.x, y: rect.y, width: w, height: rect.height }, stroke, radius, 1.0);
        self.buffer.rect(Rect { x: rect.x + rect.width - w, y: rect.y, width: w, height: rect.height }, stroke, radius, 1.0);
        Ok(())
    }

    fn fill_ellipse(&mut self, rect: Rect, fill: Rgba, opacity: f32) -> Result<(), Self::Error> {
        self.buffer.ellipse(rect, fill, opacity);
        Ok(())
    }

    fn line(&mut self, x1: f32, y1: f32, x2: f32, y2: f32, stroke: Rgba, width: f32) -> Result<(), Self::Error> {
        self.buffer.line(x1, y1, x2, y2, stroke, width);
        Ok(())
    }

    fn text(&mut self, rect: Rect, text: &str, size: f32, color: Rgba, _bold: bool) -> Result<(), Self::Error> {
        let scale = (size / 9.0).round().max(1.0) as usize;
        let x = rect.x.round() as i32;
        let y = rect.y.round() as i32;
        for (px, py) in font::rasterize(text, scale) {
            self.buffer.blend(x + px, y + py, color, 1.0);
        }
        Ok(())
    }

    fn image(&mut self, rect: Rect, _path: &str) -> Result<(), Self::Error> {
        // The CPU backend has no image decoder: draw a hatched placeholder so the
        // layout stays visible. Use the Skia or GPU backend for real image data.
        self.buffer.rect(rect, Rgba::new(224, 224, 224, 255), 0.0, 1.0);
        for i in 0..((rect.width + rect.height) / 16.0) as i32 {
            let x = rect.x + i as f32 * 16.0;
            self.buffer.line(x, rect.y, x - rect.height, rect.y + rect.height, Rgba::new(180, 180, 180, 255), 1.0);
        }
        Ok(())
    }

    fn finish(&mut self) -> Result<Vec<u8>, Self::Error> {
        Ok(self.buffer.png())
    }
}
