//! The renderer module ("penggambar"): the only part that touches pixels.
//!
//! `Caller` never draws by itself. It turns the component tree into these small
//! primitives, and the backend chosen at export time decides how they land on
//! the screen: CPU (software raster), Skia (2D GPU/CPU canvas) or GPU (wgpu).

pub mod font;
pub mod pixels;

#[cfg(feature = "skia")]
pub mod skia;
#[cfg(not(feature = "skia"))]
#[cfg(feature = "gpu")]
pub mod gpu;
#[cfg(not(feature = "skia"))]
#[cfg(not(feature = "gpu"))]
pub mod cpu;

use crate::component::{Rect, Rgba};

/// Everything a backend must be able to draw.
pub trait Renderer {
    type Error: core::fmt::Debug;

    fn begin(&mut self, width: f32, height: f32) -> Result<(), Self::Error>;
    fn fill_rect(&mut self, rect: Rect, fill: Rgba, radius: f32, opacity: f32) -> Result<(), Self::Error>;
    fn stroke_rect(&mut self, rect: Rect, stroke: Rgba, width: f32, radius: f32) -> Result<(), Self::Error>;
    fn fill_ellipse(&mut self, rect: Rect, fill: Rgba, opacity: f32) -> Result<(), Self::Error>;
    fn line(&mut self, x1: f32, y1: f32, x2: f32, y2: f32, stroke: Rgba, width: f32) -> Result<(), Self::Error>;
    fn text(&mut self, rect: Rect, text: &str, size: f32, color: Rgba, bold: bool) -> Result<(), Self::Error>;
    fn image(&mut self, rect: Rect, path: &str) -> Result<(), Self::Error>;
    /// Finish the frame and return a PNG image.
    fn finish(&mut self) -> Result<Vec<u8>, Self::Error>;
}
