//! Skia backend ("menggunakan rust dan skia"): real 2D canvas drawing.
//!
//! Enabled with the `skia` feature. `component` and `caller` never change when
//! the Skia API moves: only this file needs adjusting. Text uses the bundled
//! bitmap font, so no system font files are required.

use super::font;
use super::Renderer;
use crate::component::{Rect, Rgba};
use skia_safe::{AlphaType, Color4f, ColorType, Data, EncodedImageFormat, ImageInfo, Paint, Rect as SkRect, Surface};

pub struct SkiaRenderer {
    surface: Option<Surface>,
}

impl SkiaRenderer {
    pub fn new() -> Self {
        Self { surface: None }
    }

    fn canvas(&mut self) -> Result<&skia_safe::Canvas, String> {
        self.surface
            .as_mut()
            .map(Surface::canvas)
            .ok_or_else(|| "Skia surface is not ready".to_string())
    }
}

impl Default for SkiaRenderer {
    fn default() -> Self {
        Self::new()
    }
}

fn paint(color: Rgba, opacity: f32) -> Paint {
    let a = (f32::from(color.a) / 255.0 * opacity.clamp(0.0, 1.0)).clamp(0.0, 1.0);
    Paint::new(
        Color4f::new(f32::from(color.r) / 255.0, f32::from(color.g) / 255.0, f32::from(color.b) / 255.0, a),
        None,
    )
}

fn sk_rect(r: Rect) -> SkRect {
    SkRect::new(r.x, r.y, r.x + r.width, r.y + r.height)
}

fn radius(rect: Rect, value: f32) -> f32 {
    value.max(0.0).min(rect.width.min(rect.height) / 2.0)
}

impl Renderer for SkiaRenderer {
    type Error = String;

    fn begin(&mut self, width: f32, height: f32) -> Result<(), Self::Error> {
        let info = ImageInfo::new(
            (width.round().max(1.0) as i32, height.round().max(1.0) as i32),
            ColorType::RGBA8888,
            AlphaType::Premul,
            None,
        );
        self.surface = skia_safe::surfaces::raster(&info, None, None);
        if self.surface.is_none() {
            return Err("could not create a Skia surface".into());
        }
        self.canvas()?.clear(Color4f::new(0.0, 0.0, 0.0, 0.0));
        Ok(())
    }

    fn fill_rect(&mut self, rect: Rect, fill: Rgba, corner: f32, opacity: f32) -> Result<(), Self::Error> {
        let p = paint(fill, opacity);
        let r = radius(rect, corner);
        if r > 0.0 {
            self.canvas()?.draw_round_rect(sk_rect(rect), r, r, &p);
        } else {
            self.canvas()?.draw_rect(sk_rect(rect), &p);
        }
        Ok(())
    }

    fn stroke_rect(&mut self, rect: Rect, stroke: Rgba, width: f32, corner: f32) -> Result<(), Self::Error> {
        let mut p = paint(stroke, 1.0);
        p.set_stroke(true);
        p.set_stroke_width(width.max(1.0));
        let r = radius(rect, corner);
        if r > 0.0 {
            self.canvas()?.draw_round_rect(sk_rect(rect), r, r, &p);
        } else {
            self.canvas()?.draw_rect(sk_rect(rect), &p);
        }
        Ok(())
    }

    fn fill_ellipse(&mut self, rect: Rect, fill: Rgba, opacity: f32) -> Result<(), Self::Error> {
        self.canvas()?.draw_oval(sk_rect(rect), &paint(fill, opacity));
        Ok(())
    }

    fn line(&mut self, x1: f32, y1: f32, x2: f32, y2: f32, stroke: Rgba, width: f32) -> Result<(), Self::Error> {
        let mut p = paint(stroke, 1.0);
        p.set_stroke(true);
        p.set_stroke_width(width.max(1.0));
        self.canvas()?.draw_line((x1, y1), (x2, y2), &p);
        Ok(())
    }

    fn text(&mut self, rect: Rect, text: &str, size: f32, color: Rgba, _bold: bool) -> Result<(), Self::Error> {
        let scale = (size / 9.0).round().max(1.0) as usize;
        let p = paint(color, 1.0);
        let canvas = self.canvas()?;
        for (px, py) in font::rasterize(text, scale) {
            let x = rect.x + px as f32;
            let y = rect.y + py as f32;
            canvas.draw_rect(SkRect::new(x, y, x + 1.0, y + 1.0), &p);
        }
        Ok(())
    }

    fn image(&mut self, rect: Rect, path: &str) -> Result<(), Self::Error> {
        let data = std::fs::read(path).map_err(|e| format!("image {path}: {e}"))?;
        let mut codec = skia_safe::Codec::from_data(Data::new_copy(&data))
            .ok_or_else(|| format!("image {path}: unsupported format"))?;
        let image = codec
            .get_image(None, None)
            .map_err(|_| format!("image {path}: decode failed"))?;
        self.canvas()?.draw_image_rect(
            image,
            None,
            sk_rect(rect),
            &paint(Rgba::new(255, 255, 255, 255), 1.0),
        );
        Ok(())
    }

    #[allow(deprecated)]
    fn finish(&mut self) -> Result<Vec<u8>, Self::Error> {
        let surface = self.surface.as_mut().ok_or_else(|| "no surface to encode".to_string())?;
        let image = surface.image_snapshot();
        let data = image
            .encode_to_data(EncodedImageFormat::PNG)
            .ok_or_else(|| "PNG encoding failed".to_string())?;
        Ok(data.as_bytes().to_vec())
    }
}
